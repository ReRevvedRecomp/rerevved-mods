cmake_minimum_required(VERSION 3.25)
project(effect_plugin_checks LANGUAGES CXX)

enable_testing()

function(add_host target)
    add_executable(${target} effect_plugins_host.cpp)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON ENABLE_EXPORTS ON)
    target_include_directories(${target} PRIVATE "${SDK_DIR}/include" "${MODS_ROOT}/src/common/api")
    if(UNIX)
        target_link_libraries(${target} PRIVATE ${CMAKE_DL_LIBS})
    endif()
endfunction()

add_host(effect_plugins_host)
foreach(missing VERSION SUPPORT REGISTRATION)
    string(TOLOWER ${missing} missing_name)
    add_host(missing_${missing_name}_host)
    target_compile_definitions(missing_${missing_name}_host PRIVATE OMIT_${missing})
endforeach()

add_host(title_effect_plugins_host)
target_sources(title_effect_plugins_host PRIVATE "${TITLE_DIR}/src/building_wonder_effect_rules.cpp")
target_compile_definitions(title_effect_plugins_host PRIVATE USE_TITLE_API BUILDING_EFFECT_RULES_API_EXPORTS WONDER_EFFECT_RULES_API_EXPORTS)

if(WIN32)
    set(prefix "")
    set(extension ".dll")
elseif(APPLE)
    set(prefix "lib")
    set(extension ".dylib")
else()
    set(prefix "lib")
    set(extension ".so")
endif()
set(wonder "${MODS_ROOT}/mods/stonehenge-religion/code/${RUNTIME_PLATFORM}/${prefix}stonehenge_religion${extension}")
set(building "${MODS_ROOT}/mods/chinese-library-gold/code/${RUNTIME_PLATFORM}/${prefix}chinese_library_gold${extension}")

function(add_plugin_check name host mode expression)
    add_test(NAME ${name} COMMAND ${host} ${mode} ${ARGN})
    set_tests_properties(${name} PROPERTIES PASS_REGULAR_EXPRESSION "${expression}" FAIL_REGULAR_EXPRESSION "Host check failed")
endfunction()

foreach(kind wonder building)
    foreach(missing VERSION SUPPORT REGISTRATION)
        string(TOLOWER ${missing} missing_name)
        add_plugin_check(${kind}.missing_${missing_name} missing_${missing_name}_host missing "required title API exports are missing" ${kind} "${${kind}}")
    endforeach()
    add_plugin_check(${kind}.abi_mismatch effect_plugins_host abi-mismatch "incompatible title API" ${kind} "${${kind}}")
    add_plugin_check(${kind}.support_error effect_plugins_host support-error "support query failed" ${kind} "${${kind}}")
    foreach(mode support-small support-invalid)
        add_plugin_check(${kind}.${mode} effect_plugins_host ${mode} "invalid runtime support response" ${kind} "${${kind}}")
    endforeach()
    add_plugin_check(${kind}.unavailable effect_plugins_host unavailable "integration unavailable; no rule registered" ${kind} "${${kind}}")
    foreach(mode registration-error registration-unavailable)
        add_plugin_check(${kind}.${mode} effect_plugins_host ${mode} "registration rejected" ${kind} "${${kind}}")
    endforeach()
    add_plugin_check(${kind}.success effect_plugins_host success "rule registered" ${kind} "${${kind}}")
    add_plugin_check(${kind}.pinned_title title_effect_plugins_host unavailable "integration unavailable; no rule registered" ${kind} "${${kind}}")
endforeach()

foreach(mode success unavailable)
    add_plugin_check(both.${mode}.forward effect_plugins_host ${mode} "Chinese Library Gold:" wonder "${wonder}" building "${building}")
    add_plugin_check(both.${mode}.reverse effect_plugins_host ${mode} "Stonehenge Religion:" building "${building}" wonder "${wonder}")
endforeach()
add_plugin_check(both.pinned_title.forward title_effect_plugins_host unavailable "Chinese Library Gold:" wonder "${wonder}" building "${building}")
add_plugin_check(both.pinned_title.reverse title_effect_plugins_host unavailable "Stonehenge Religion:" building "${building}" wonder "${wonder}")

foreach(kind hills-production cataphracts-defense hoplite-loyalty jaguar-woodsman keshik-movement mongol-horseback)
    string(REPLACE "-" "_" stem "${kind}")
    set(plugin "${MODS_ROOT}/mods/${kind}/code/${RUNTIME_PLATFORM}/${prefix}${stem}${extension}")
    set(modes success duplicate internal invalid)
    if(NOT kind STREQUAL "hills-production")
        list(APPEND modes gameplay-first)
    endif()
    if(kind STREQUAL "hoplite-loyalty" OR kind STREQUAL "jaguar-woodsman")
        list(APPEND modes gameplay-second)
    endif()
    foreach(mode IN LISTS modes)
        add_test(NAME registration.${kind}.${mode}
            COMMAND "${CMAKE_COMMAND}" "-DHOST=$<TARGET_FILE:registration_plugins_host>"
                "-DKIND=${kind}" "-DMODE=${mode}" "-DPLUGIN=${plugin}"
                -P "${CMAKE_CURRENT_SOURCE_DIR}/check_registration_plugin.cmake")
    endforeach()
endforeach()
